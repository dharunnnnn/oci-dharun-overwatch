:INPUT ACCEPT 
:FORWARD ACCEPT 
:OUTPUT ACCEPT 
:DOCKER - 
:DOCKER-BRIDGE - 
:DOCKER-CT - 
:DOCKER-FORWARD - 
:DOCKER-INTERNAL - 
:DOCKER-USER -

-A FORWARD -j DOCKER-FORWARD
-A DOCKER-FORWARD -j DOCKER-CT
-A DOCKER-FORWARD -j DOCKER-INTERNAL

:PREROUTING ACCEPT 
:INPUT ACCEPT 
:OUTPUT ACCEPT 
:POSTROUTING ACCEPT 
:DOCKER -
